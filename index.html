Backend Cloud • PostgreSQL + RLS

Configurar Tablas en Supabase (10 segundos)

  1. Entrá a tu proyecto en Supabase → SQL Editor → New query.
  2. Pegá el siguiente código y presioná Run para crear miner_profiles y withdrawals con seguridad por usuario.
  3. Activá Authentication → Providers → Google y pegá tu SUPABASE_URL y SUPABASE_ANON_KEY al inicio del <script> de este archivo.
schema_supabase.sql 
-- 1. Tabla de perfil y estado del ciclo de minado por usuario
create table if not exists public.miner_profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  email text,
  usdt_balance numeric(18, 6) not null default 0,
  total_mined numeric(18, 6) not null default 0,
  completed_cycles integer not null default 0,
  cycle_start_time timestamptz,
  cycle_duration_hours integer not null default 1,
  upgrades jsonb not null default '{"optimism_node":0,"antminer_s21":0,"immersion_cooling":0,"direct_usdt_pool":0}'::jsonb,
  default_network text not null default 'OPTIMISM',
  default_wallet text not null default '',
  updated_at timestamptz not null default now()
);

-- 2. Tabla de historial de retiros hacia Lemon
create table if not exists public.withdrawals (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  network text not null,
  wallet_address text not null,
  amount_usdt numeric(18, 6) not null,
  status text not null default 'Pendiente de envío',
  created_at timestamptz not null default now()
);

-- 3. Habilitar Row Level Security (RLS) para proteger tus datos
alter table public.miner_profiles enable row level security;
alter table public.withdrawals enable row level security;

create policy "Usuario gestiona su propio perfil"
  on public.miner_profiles for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);

create policy "Usuario gestiona sus propios retiros"
  on public.withdrawals for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
